  const speakAndGoNext = (sceneIndex) => {

    const text = scenes[sceneIndex].script;
    const utterance = new SpeechSynthesisUtterance(text);
    
    // 音声の設定（日本語、少し高め・少し早めで子ども向けに）
    utterance.lang = 'ja-JP';
    utterance.pitch = 1.2;
    utterance.rate = 1.1;

    utterance.onstart = () => {
      setIsSpeaking(true);
    };

    utterance.onend = () => {
      setIsSpeaking(false);
      // 自動再生中で、かつ最後のシーンでなければ次へ進む
      if (isAutoPlayingRef.current) {
        if (sceneIndex < scenes.length - 1) {
          // 少し間をあけて次のシーンへ
          setTimeout(() => {
            setCurrentScene(prev => {
              const next = prev + 1;
              speakAndGoNext(next);
              return next;
            });
          }, 1000); // 1秒の間
        } else {
          // 最後まで終わったら自動再生終了
          setIsAutoPlaying(false);
          isAutoPlayingRef.current = false;
          // 録画中なら録画も停止して保存処理へ
          if (mediaRecorderRef.current && mediaRecorderRef.current.state === "recording") {
            mediaRecorderRef.current.stop();
          }
        }
      }
    };

    utterance.onerror = (e) => {
      console.error("Speech error", e);
      setIsSpeaking(false);
    };

    synthRef.current.speak(utterance);
  };

      // 停止する
      setIsAutoPlaying(false);
      synthRef.current.cancel();

      if (mediaRecorderRef.current && mediaRecorderRef.current.state === "recording") {
        mediaRecorderRef.current.stop();
      }
    } else {
      // 自動再生スタート
      setCurrentScene(0);
      setIsAutoPlaying(true);
      // ブラウザの制限により、ユーザーアクション直後に呼び出す必要がある
      setTimeout(() => {
         speakAndGoNext(0);

  // コンポーネント破棄時に音声を止める
  useEffect(() => {
    return () => {
      if (synthRef.current) {
        synthRef.current.cancel();
      }
      if (mediaRecorderRef.current && mediaRecorderRef.current.state === "recording") {
        mediaRecorderRef.current.stop();
      }
    };
  }, []);

  // ブラウザの録画機能を使って動画を作成・保存するロジック
  const startRecordingAndPlay = async () => {
    try {
      // 画面共有と音声の取得（ユーザーに許可を求めるダイアログが出ます）
      const stream = await navigator.mediaDevices.getDisplayMedia({
        video: { displaySurface: "browser" },
        audio: true // ※重要: これによりシステム（タブ）音声を取得します
      });

      // 録画データの保存先
      recordedChunksRef.current = [];
      const options = { mimeType: 'video/webm' };
      let recorder;
      try {
        recorder = new MediaRecorder(stream, options);
      } catch (e) {
        // オプション非対応時のフォールバック
        recorder = new MediaRecorder(stream);
      }

      mediaRecorderRef.current = recorder;

      // データが生成されるたびにチャンクに追加
      recorder.ondataavailable = (e) => {
        if (e.data.size > 0) {
          recordedChunksRef.current.push(e.data);
        }
      };

      // 録画停止時（最後まで再生が終わった時）のファイル化とダウンロード処理
      recorder.onstop = () => {
        const blob = new Blob(recordedChunksRef.current, {
          type: recorder.mimeType || 'video/webm'
        });
        const url = URL.createObjectURL(blob);
        const a = document.createElement('a');
        document.body.appendChild(a);
        a.style = 'display: none';
        a.href = url;
        a.download = 'TableTennisHistory.webm'; // WebM形式でダウンロード
        a.click();
        window.URL.revokeObjectURL(url);

        // カメラ/マイクの使用（画面共有）を完全に停止
        stream.getTracks().forEach(track => track.stop());
        setIsRecording(false);
        setIsAutoPlaying(false);
        synthRef.current.cancel();
      };

        {isSpeaking && (

          <div className="flex-1 flex items-center justify-center">
